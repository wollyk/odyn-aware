# AuroraView object-tracker sidecar.
#
# Pulls latest.jpg from Frigate at ~5Hz per active camera, runs
# YOLOv8s + ByteTrack, and broadcasts per-camera tracker state over
# a FastAPI websocket.
#
# Why these versions:
#   - ultralytics 8.3+: stable .track(persist=True) tracker integration.
#   - httpx: native async client for snapshot polling. Avoids blocking
#     the event loop while uvicorn serves websockets.
#   - websockets: required by uvicorn for WS support (uvicorn[standard]
#     would also bring it but we avoid the heavyweight extra deps).
fastapi
uvicorn
websockets
httpx
pillow
# `lap` is required by ByteTrack's Hungarian-assignment step. Ultralytics
# treats it as optional (only needed when `model.track(tracker="bytetrack")`
# is called), so it's missing from the default install. Without it the
# very first track call throws "No module named 'lap'" on every frame.
lap>=0.5.0
ultralytics>=8.3.0,<9.0
